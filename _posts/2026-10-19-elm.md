<script>
  (async () => {
    const status = document.getElementById('elm-chart-status');
    if (!status) return;

    try {
      const response = await fetch('/assets/elm.csv');
      if (!response.ok) throw new Error(`Request failed (${response.status})`);

      const csv = await response.text();
      const list = csv.trim().split(/\r?\n/).slice(1)
        .filter((line) => line.trim())
        .map((line) => {
          const [time, cost] = line.split(',');
          return { time: time.trim(), cost: Number(cost) };
        });

      const trace1 = {
        x: list.map((row) => row.time),
        y: list.map((row) => row.cost),
        mode: 'lines+markers',
        name: 'Cost(overall)',
        line: {shape: 'spline', smoothing: 0.5},

      connectgaps: true,
        type: 'scatter'
      };

      var trace2 = {
        x: list.map((row) => row.time),
        y: list.map((row,i) => i==0? 0:(row.cost - list[i-1].cost) ),
        name: 'Cost(hourly)',
        yaxis: 'y2',
        type: 'bar',
        width: 50*60000, //50min in miliseconds
      };

      var layout = {
        yaxis: {
          overlaying: 'y2',
        },
        yaxis2: {
          side: 'right'
        }
      };


      await Plotly.newPlot('elm-chart', [trace1,trace2],layout);


      status.textContent = `ELM: ${list.length} data points`;
    } catch (error) {
      status.textContent = `Unable to load chart: ${error.message}`;
    }
  })();
</script>



<script>
  (async () => {
    const status = document.getElementById('copilot-chart-status');
    if (!status) return;

    try {
      const response = await fetch('/assets/copilot.csv');
      if (!response.ok) throw new Error(`Request failed (${response.status})`);

      const csv = await response.text();
      const models = new Set()
      const list = csv.trim().split(/\r?\n/).slice(1)
        .filter((line) => line.trim())
        .map((line) => {
          const [time, model, grossquantity] = line.split(',');
          m=model.trim()
          models.add(m)
          return { time: time.trim(), model: m, cost: Number(grossquantity) };
        });

      const traces = Array.from(models).map(m=>{
        let l = list.filter(l=>l.model == m)
        return {
        x: l.map((row) => row.time),
        y: l.map((row) => row.cost ),
        name: m,
        yaxis: 'y2',
        type: 'bar',
        width: 60*60000*23, //50min in miliseconds
        }
      })

      const dailyCosts = new Map();
      list.forEach((row) => {
        const date = row.time.slice(0, 10);
        dailyCosts.set(date, (dailyCosts.get(date) || 0) + row.cost);
      });

      const x = Array.from(dailyCosts.keys()).sort();
      let cumulativeCost = 0;
      let currentMonth = '';
      const cumulativeX = [];
      const y = [];
      x.forEach((date) => {
        const month = date.slice(0, 7);
        if (month !== currentMonth) {
          if (currentMonth) {
            cumulativeX.push(date);
            y.push(null);
          }
          cumulativeCost = 0;
          currentMonth = month;
        }
        cumulativeCost += dailyCosts.get(date);
        cumulativeX.push(date);
        y.push(cumulativeCost);
      });

      const trace1 = {
        x: cumulativeX,
        y: y,
        mode: 'lines+markers',
        name: 'Cost(monthly)',
        line: {shape: 'spline', smoothing: 0.5},

        connectgaps: false,
        type: 'scatter',
        yaxis: 'y',
      };



      var layout = {
        yaxis: {
          rangemode: 'tozero',
          overlaying: 'y2',
        },
        yaxis2: {
          side: 'right'
        },
        barmode: 'stack',
      };




      await Plotly.newPlot('copilot-chart', [trace1, ...traces, ],layout);


      status.textContent = `Copilot Usage.`;
    } catch (error) {
      status.textContent = `Unable to load chart: ${error.message}`;
    }
  })();
</script>
